
def objective(trial) -> float:
    scale  = trial.suggest_float("scale", 0.15, 2.5)   
    thresh = std_e * scale                              
    e_v, r_v = e_arr[:val_end], ret_arr[:val_end]      
    sigs = np.where(e_v > thresh, 1.0,
            np.where(e_v < -thresh, -1.0, 0.0))
    changes = np.abs(np.diff(np.r_[0.0, sigs]))
    changes[-1] += abs(sigs[-1])
    strat = sigs * r_v - fee_rate * changes
    return evaluate_objective(strat)
