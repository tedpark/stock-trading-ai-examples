

def forward_proba_causal(self, features: np.ndarray) -> None:
    Xs = self._scaler.transform(features)

    
    log_e  = self.model._compute_log_likelihood(Xs)
    log_e -= log_e.max(axis=1, keepdims=True)
    e      = np.exp(log_e)

    # α_1 = π ⊙ e_1
    alpha = self.model.startprob_ * e[0]
    alpha = alpha / alpha.sum()

    
    for t in range(1, len(Xs)):
        alpha = (alpha @ self.model.transmat_) * e[t]
        alpha = alpha / alpha.sum()

    self._alpha = alpha
