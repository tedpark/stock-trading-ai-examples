def objective(trial):
    vote_thresh = trial.suggest_float("vote_thresh", 0.50, 0.85)
    signals = signals_from_votes(vote_thresh)
    changes = np.abs(np.diff(np.r_[0.0, signals]))
    changes[-1] += abs(signals[-1])  # Final liquidation.
    strat = signals * next_returns - fee_rate * changes
    downside = np.sqrt(np.mean(np.minimum(strat, 0.0) ** 2))
    if downside == 0:
        return -float("inf")  # Undefined Sortino: reject this candidate.
    sortino = strat.mean() / downside * annualization_factor
    turnover = float(changes.mean())
    turn_penalty = max(0, (turnover - 0.05) * 3)
    return sortino - turn_penalty
