def get_account_summary(self, account: str) -> Dict[str, Any]:
    from math import isfinite
    if not account or not self._connected:
        raise ValueError("connected broker and explicit account required")
    tags = {"NetLiquidation", "TotalCashValue", "AvailableFunds",
            "GrossPositionValue", "Leverage", "MaintMarginReq"}
    rows = self._run(self._ib.accountSummaryAsync(account), timeout=10)
    result = {}
    for row in rows:
        if row.account != account or row.tag not in tags:
            continue
        value = float(row.value)
        if not isfinite(value):
            raise ValueError("nonfinite account value")
        # Preserve reported currency; do not silently assume USD.
        result.setdefault(row.tag, {})[row.currency] = value
    if "NetLiquidation" not in result:
        raise ValueError("account summary is not ready")
    return result
