import numpy as np
import statsmodels.api as sm
from statsmodels.stats.multitest import multipletests
from statsmodels.tsa.stattools import coint

def find_cointegrated_pairs(self, symbols) -> list:
    tested = []
    for sector, sector_symbols in SECTOR_SYMBOLS.items():
        prices = self._load_prices(sector_symbols, period="3y")
        corr_matrix = prices.pct_change().dropna().corr()

        for i, sym_a in enumerate(corr_matrix.columns):
            for sym_b in corr_matrix.columns[i+1:]:
                
                corr = corr_matrix.loc[sym_a, sym_b]
                if corr < self.min_correlation:
                    continue

                pair = prices[[sym_a, sym_b]].dropna()
                price_a = pair[sym_a]
                price_b = pair[sym_b]

                
                _, pvalue, _ = coint(price_a.values, price_b.values)
                X = sm.add_constant(price_b.to_numpy())
                fit = sm.OLS(price_a.to_numpy(), X).fit()
                intercept, beta = fit.params
                spread = price_a - (intercept + beta * price_b)
                tested.append({
                    "symbols": (sym_a, sym_b), "corr": corr,
                    "pvalue": pvalue, "beta": beta,
                    "halflife": calc_halflife(spread.to_numpy()),
                })

    if not tested:
        return []

    
    reject, qvalues, _, _ = multipletests(
        [x["pvalue"] for x in tested], alpha=self.coint_pvalue, method="fdr_bh"
    )
    selected = []
    for row, keep, qvalue in zip(tested, reject, qvalues):
        hl = row["halflife"]
        if keep and np.isfinite(hl) and hl <= self.max_halflife:
            selected.append({**row, "qvalue": qvalue})
    return sorted(selected, key=lambda x: (x["qvalue"], x["halflife"]))
