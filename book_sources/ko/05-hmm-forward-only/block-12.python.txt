def step_forward(self, feat_row: np.ndarray) -> np.ndarray | None:
    """Filter one observation using full covariance matrices in log space."""
    from scipy.special import logsumexp
    from scipy.stats import multivariate_normal

    x = self._scaler.transform(np.asarray(feat_row).reshape(1, -1))[0]
    log_e = np.array([
        multivariate_normal.logpdf(x, mean=mean, cov=cov)
        for mean, cov in zip(self.model.means_, self.model.covars_)
    ])
    predicted = self._alpha @ self.model.transmat_
    with np.errstate(divide="ignore"):
        log_alpha = np.log(predicted) + log_e
    normalizer = logsumexp(log_alpha)
    if not np.isfinite(normalizer):
        return None  # Preserve the previous state on invalid evidence.
    self._alpha = np.exp(log_alpha - normalizer)
    return self._to_logical(self._alpha)
