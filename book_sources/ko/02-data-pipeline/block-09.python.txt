def compute_zscore(close, window=20):
    T = len(close)
    z = np.full(T, np.nan, dtype=np.float64)
    for i in range(window, T):
        win = close[i - window: i]
        mu = float(np.mean(win))
        sigma = float(np.std(win))
        if sigma > 1e-10:
            z[i] = (close[i] - mu) / sigma
    return z
