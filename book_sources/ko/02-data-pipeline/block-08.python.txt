def download_ohlcv(symbol, interval="1d"):
    # Request flat columns for one ticker (current yfinance API).
    df = yf.download(symbol, period="5y", interval=interval,
                     auto_adjust=True, multi_level_index=False)
    if df.empty:
        raise ValueError("No bars returned")
    df = df.copy()
    # Also accept the single-ticker MultiIndex form from a wrapper/cache.
    if isinstance(df.columns, pd.MultiIndex):
        if df.columns.nlevels != 2:
            raise ValueError("Expected Price/Ticker column levels")
        tickers = df.columns.get_level_values(1).unique()
        if len(tickers) != 1 or tickers[0] != symbol:
            raise ValueError("Expected only the requested ticker")
        df.columns = df.columns.get_level_values(0)
    df.columns = [str(c).lower() for c in df.columns]
    required = ["open", "high", "low", "close", "volume"]
    if df.columns.duplicated().any() or not set(required) <= set(df.columns):
        raise ValueError("Missing or duplicate OHLCV columns")
    idx = pd.to_datetime(df.index)
    if idx.hasnans or idx.has_duplicates or not idx.is_monotonic_increasing:
        raise ValueError("Expected unique, ordered bar timestamps")
    if idx.tz is not None:
        idx = idx.tz_convert("UTC").tz_localize(None)
    df.index = idx
    return df[required]
