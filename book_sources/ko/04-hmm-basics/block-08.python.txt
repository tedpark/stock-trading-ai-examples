def _update_hmm(self, bar_idx: int) -> None:
    for sym in self._symbols:
        hmm = self._hmms[sym]

        need_fit = (hmm.model is None
                    or bar_idx - hmm._last_fit_idx >= HMM_REFIT_DAYS)

        if need_fit and bar_idx >= HMM_MIN_DAYS:
            feat_train = self._features[sym][:bar_idx]      # 인과적 슬라이스
            if hmm.fit(feat_train):
                hmm._last_fit_idx = bar_idx

        if hmm.model is not None:
            recent = self._features[sym][max(0, bar_idx - 30): bar_idx]
            probs = hmm.predict_proba(recent)
            if probs is not None and len(probs) > 0:
                self._regime_probs[sym] = probs[-1]         # 최신 봉의 확률 벡터
