class KalmanFilter1D:
    """단변량 1-D 칼만 필터 (피처 평활화용)."""

    def __init__(self, Q: float = 1e-4, R: float = 1e-2):
        self.Q, self.R = Q, R
        self.x = 0.0   # 상태 추정치
        self.P = 1.0   # 추정치 분산

    def update(self, z: float) -> float:
        self.P += self.Q                 # 1단계: 예측 (분산 증가)
        K = self.P / (self.P + self.R)  # 칼만 이득
        self.x += K * (z - self.x)      # 2단계: 상태 갱신
        self.P *= 1 - K                  # 분산 축소
        return self.x
