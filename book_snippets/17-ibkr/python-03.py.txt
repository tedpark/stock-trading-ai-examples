async def _get_session_info_async(self, symbol: str) -> dict[str, str]:
    from ib_async import Stock

    contract = Stock(symbol, "SMART", "USD")
    details = await self._ib.reqContractDetailsAsync(contract)
    if not details:
        raise ValueError(f"Contract details not found: {symbol}")

    cd = details[0]
    return {
        "symbol": symbol,
        "timezone": cd.timeZoneId,
        "trading_hours": cd.tradingHours,
        "liquid_hours": cd.liquidHours,
    }
