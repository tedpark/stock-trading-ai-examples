def get_positions(self, account: str) -> List[Dict[str, Any]]:
    if not account:
        raise ValueError("explicit account required")
    with self._lock:
        if not self._connected:
            raise ConnectionError("positions unavailable while disconnected")
        # _run belongs to the separate broker-loop adapter described above.
        # A timeout propagates; it must not become a confirmed empty account.
        rows = self._run(self._ib.reqPositionsAsync(), timeout=8)
        return [
            {"account": p.account, "con_id": p.contract.conId,
             "symbol": p.contract.symbol, "size": p.position,
             "avg_cost": p.avgCost, "market_price": None}
            for p in rows if p.account == account and p.position != 0
        ]
